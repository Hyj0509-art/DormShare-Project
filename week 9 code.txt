#include "esp_camera.h"
#include <WiFi.h>
#include <WebServer.h>

// ========== WiFi Information ==========
const char* ssid = "HONOR 90";
const char* password = "hyj050905";

// ========== Camera Pins ==========
#define PWDN_GPIO_NUM     -1
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM      15
#define SIOD_GPIO_NUM       4
#define SIOC_GPIO_NUM       5
#define Y9_GPIO_NUM        16
#define Y8_GPIO_NUM        17
#define Y7_GPIO_NUM        18
#define Y6_GPIO_NUM        12
#define Y5_GPIO_NUM        10
#define Y4_GPIO_NUM         8
#define Y3_GPIO_NUM         9
#define Y2_GPIO_NUM        11
#define VSYNC_GPIO_NUM     6
#define HREF_GPIO_NUM      7
#define PCLK_GPIO_NUM      13

// ========== User Pins ==========
#define PIN_BUZZER        1
#define PIN_BUTTON        0
#define PIN_LED           14

WebServer server(80);

// State variables
bool deviceState = true;      // true=ON, false=OFF
bool lastButtonState = HIGH;  // Last button state
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 200;  // Debounce delay

// Minimal web page
const char index_html[] PROGMEM = R"rawliteral(

// Update LED and buzzer state
void updateOutputs() {
    if (deviceState) {
        digitalWrite(PIN_LED, HIGH);   // LED ON
        digitalWrite(PIN_BUZZER, HIGH); // Buzzer ON
    } else {
        digitalWrite(PIN_LED, LOW);    // LED OFF
        digitalWrite(PIN_BUZZER, LOW);  // Buzzer OFF
    }
}

// Check button (toggle mode)
void checkButton() {
    int currentState = digitalRead(PIN_BUTTON);
    
    // Detect button press (HIGH to LOW transition)
    if (currentState == LOW && lastButtonState == HIGH) {
        // Debounce handling
        if ((millis() - lastDebounceTime) > debounceDelay) {
            lastDebounceTime = millis();
            // Toggle state
            deviceState = !deviceState;
            updateOutputs();
            Serial.print("Button pressed, state toggled to: ");
            Serial.println(deviceState ? "ON" : "OFF");
        }
    }
    
    lastButtonState = currentState;
}

// Camera initialization
void initCamera() {
    camera_config_t config;
    config.pin_pwdn = PWDN_GPIO_NUM;
    config.pin_reset = RESET_GPIO_NUM;
    config.pin_xclk = XCLK_GPIO_NUM;
    config.pin_sscb_sda = SIOD_GPIO_NUM;
    config.pin_sscb_scl = SIOC_GPIO_NUM;
    config.pin_d7 = Y9_GPIO_NUM;
    config.pin_d6 = Y8_GPIO_NUM;
    config.pin_d5 = Y7_GPIO_NUM;
    config.pin_d4 = Y6_GPIO_NUM;
    config.pin_d3 = Y5_GPIO_NUM;
    config.pin_d2 = Y4_GPIO_NUM;
    config.pin_d1 = Y3_GPIO_NUM;
    config.pin_d0 = Y2_GPIO_NUM;
    config.pin_vsync = VSYNC_GPIO_NUM;
    config.pin_href = HREF_GPIO_NUM;
    config.pin_pclk = PCLK_GPIO_NUM;

    config.xclk_freq_hz = 10000000;
    config.pixel_format = PIXFORMAT_JPEG;
    config.frame_size = FRAMESIZE_QVGA;
    config.jpeg_quality = 12;
    config.fb_count = 1;

    esp_err_t err = esp_camera_init(&config);
    if (err != ESP_OK) {
        Serial.printf("Camera init failed: 0x%x\n", err);
        return;
    }
    Serial.println("✅ Camera initialized successfully");
}

// Video stream
void handleStream() {
    WiFiClient client = server.client();
    client.print("HTTP/1.1 200 OK\r\nContent-Type: multipart/x-mixed-replace; boundary=frame\r\n\r\n");

    while (client.connected()) {
        checkButton();  // Real-time button detection
        
        camera_fb_t *fb = esp_camera_fb_get();
        if (!fb) continue;

        client.print("--frame\r\nContent-Type: image/jpeg\r\n\r\n");
        client.write(fb->buf, fb->len);
        client.print("\r\n");
        esp_camera_fb_return(fb);

        delay(30);
    }
}

void handleRoot() {
    server.send(200, "text/html", index_html);
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("\nStarting ESP32-CAM...");

    pinMode(PIN_BUTTON, INPUT_PULLUP);
    pinMode(PIN_LED, OUTPUT);
    pinMode(PIN_BUZZER, OUTPUT);

    // Initial state: ON
    deviceState = true;
    updateOutputs();

    initCamera();

    WiFi.begin(ssid, password);
    Serial.print("Connecting to WiFi");
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("\n✅ WiFi Connected!");
    Serial.print("IP Address: http://");
    Serial.println(WiFi.localIP());
    Serial.println("Press BOOT button to toggle LED and buzzer state");

    server.on("/", handleRoot);
    server.on("/stream", handleStream);
    server.begin();
}

void loop() {
    checkButton();      // Check button for state toggle
    server.handleClient();
}
